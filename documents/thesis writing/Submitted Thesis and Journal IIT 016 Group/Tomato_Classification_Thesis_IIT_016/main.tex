\documentclass[a4paper, 12pt]{article}

%----------------------- Packages -------------------------------------
% Package list matches the official UWU Thesis LaTeX Template, plus
% textcomp (required for \textdegree, used in Chapters 2 and 3).

\usepackage{mathptmx}          % Times New Roman for text and maths
\usepackage{setspace}
\usepackage{geometry}
\usepackage{titlesec}
\usepackage{graphicx}
\usepackage{multirow}

\usepackage{longtable}

\usepackage{float}

\usepackage{datetime}
\usepackage{natbib}
\usepackage{subcaption}        % also provides \captionsetup, via caption
\usepackage{hyperref}
\usepackage[acronym, toc]{glossaries}

\newdateformat{monthyeardate}{\monthname[\THEMONTH] \THEYEAR}
\makeglossaries

%----------------------- Referencing ----------------------------------
% Harvard / Author-Year system, as required by guideline H:2(7)A.
\bibliographystyle{agsm}

%----------------------- Page geometry --------------------------------
\geometry{left=3.85cm, right=2.5cm, top=2.5cm, bottom=2.5cm}
\setstretch{1.5}
\setlength{\parindent}{0pt}
\setlength{\parskip}{1em plus 0.2em minus 0.2em}  % one blank line between paragraphs

%----------------------- Table and figure captions --------------------
\captionsetup[table]{labelsep=colon, singlelinecheck=false,
                     justification=raggedright, font=normalsize}
\captionsetup[figure]{labelsep=colon, singlelinecheck=false,
                      justification=raggedright, font=normalsize}
\renewcommand{\arraystretch}{1.15}

% Chapter-based numbering, so items read as Table 3.1, Figure 3.2,
% Equation (3.4) -- matching the List of Figures and List of Tables.
\numberwithin{table}{section}
\numberwithin{figure}{section}
\numberwithin{equation}{section}

% Tables are set single spaced and one size down so that wide tables fit
% the 14.65 cm text block; body text stays at 12 pt / 1.5 spacing. The hook
% is applied to the float, not to tabular, because tabularx measures its X
% columns over several passes and must not see the font change mid-way.
\usepackage{etoolbox}
\AtBeginEnvironment{table}{\setstretch{1.0}\small}

%----------------------- Heading formats ------------------------------
\titleformat{\subsection}{\normalfont\bfseries}{\thesubsection}{1em}{}
\titleformat{\subsubsection}{\normalfont\itshape}{\thesubsubsection}{1em}{}
\titlespacing*{\section}{0pt}{20pt plus 4pt minus 2pt}{20pt plus 2pt minus 2pt}
